# Ledger Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an independent, hand-entered auditor cash book (sidebar Ledger) with two wallets, calculated remainings, and a weekly owner remit that may be partial.

**Architecture:** New `ledger_entries` table only. Pure balance math in `ledgerUtils`. `ledgerService` reads and writes that table and nothing else. Vue page + modal follow Expenses layout; route and sidebar are Admin + Auditor.

**Tech Stack:** Vue 3, Vue Router, Supabase JS, Vitest (`npm test` / `npx vitest run <file>`).

**Spec:** `docs/superpowers/specs/2026-09-25-ledger-design.md`

## Global Constraints

- Ledger is independent data: the auditor types every amount; do not read, join, or sync `orders`, `order_items`, `expenses`, `store_sessions`, `sales_summary`, inventory, or users.
- Migration is add-only: `CREATE TABLE ledger_entries` plus indexes, checks, and RLS on that table. No `ALTER` or `DROP` on existing tables.
- `ledgerService` queries only `ledger_entries`. `created_by` is plain text (current username), not a foreign key.
- Currency is PHP. Calendar is Asia/Manila. A week is Monday 00:00 through Sunday 23:59 Manila time.
- Amounts are stored positive; sign comes from `type` + `wallet`. Wallet remaining is the sum of **all** entries (not the filtered table).
- Owner remit is never forced to equal remaining. Remitting less than remaining is the normal weekly case.
- Tests are written before production code for each unit. Do not change Expenses, Orders, or store-session behavior.
- Follow existing test style: Vitest, `vi.mock` for supabase/services, `createApp` + `document.createElement` for Vue components.

## Review Focus

These are the cases most likely to bite the auditor if missed. Each is pinned to a task test below.

1. **Cards use all entries, table uses the date filter.** After week 1 leftover float, a “this week” filter must not zero the remaining cards. Task 6.
2. **Cash to POS with no starting cash is blocked.** First outflow cannot go negative. Task 1.
3. **Partial owner remit is allowed and leaves leftover.** Remit-to-owner must not require the full remaining. Tasks 1 and 6.
4. **Petty-issue edit/delete is a pair.** Updating or deleting one row must touch both rows with the same `transfer_id`. Task 3.
5. **Service never touches other business tables.** `list`/`create` call `supabase.from('ledger_entries')` only; source file must not mention `orders` or `expenses`. Task 3.

import type {
  CreateLedgerEntryInput,
  LedgerEntry,
  LedgerEntryType,
  LedgerValidationResult,
  LedgerWallet
} from '../../models/Ledger'

export function entryDelta(entry: Pick<LedgerEntry, 'type' | 'wallet' | 'amount'>): {
  auditor_cash: number
  petty_cash: number
} {
  const amount = entry.amount
  const zero = { auditor_cash: 0, petty_cash: 0 }
  if (entry.type === 'opening_cash' || entry.type === 'pos_remittance') {
    return { ...zero, auditor_cash: amount }
  }
  if (
    entry.type === 'pos_float' ||
    entry.type === 'owner_remittance' ||
    (entry.type === 'expense' && entry.wallet === 'auditor_cash') ||
    (entry.type === 'petty_issue' && entry.wallet === 'auditor_cash')
  ) {
    return { ...zero, auditor_cash: -amount }
  }
  if (entry.type === 'petty_issue' && entry.wallet === 'petty_cash') {
    return { ...zero, petty_cash: amount }
  }
  if (entry.type === 'expense' && entry.wallet === 'petty_cash') {
    return { ...zero, petty_cash: -amount }
  }
  return zero
}

export function walletRemaining(entries: LedgerEntry[], wallet: LedgerWallet): number {
  return entries.reduce((sum, entry) => sum + entryDelta(entry)[wallet], 0)
}

export function dailyRemaining(
  entries: LedgerEntry[],
  wallet: LedgerWallet,
  date: string
): number {
  return walletRemaining(entries.filter((entry) => entry.entry_date <= date), wallet)
}

export function manilaYmd(date: Date): string {
  return new Intl.DateTimeFormat('en-CA', {
    timeZone: 'Asia/Manila',
    year: 'numeric',
    month: '2-digit',
    day: '2-digit'
  }).format(date)
}

function addDaysToYmd(ymd: string, days: number): string {
  const [year, month, day] = ymd.split('-').map(Number)
  const utc = new Date(Date.UTC(year, month - 1, day + days))
  return utc.toISOString().slice(0, 10)
}

export function manilaWeekWindow(now: Date = new Date()): { start: string; end: string } {
  const ymd = manilaYmd(now)
  const weekday = new Intl.DateTimeFormat('en-US', {
    timeZone: 'Asia/Manila',
    weekday: 'short'
  }).format(now)
  const dayOffset: Record<string, number> = {
    Sun: 6, Mon: 0, Tue: 1, Wed: 2, Thu: 3, Fri: 4, Sat: 5
  }
  const start = addDaysToYmd(ymd, -(dayOffset[weekday] ?? 0))
  return { start, end: addDaysToYmd(start, 6) }
}

export function entriesInDateRange(
  entries: LedgerEntry[],
  start: string,
  end: string
): LedgerEntry[] {
  return entries.filter((entry) => entry.entry_date >= start && entry.entry_date <= end)
}

export function weekSummary(
  entries: LedgerEntry[],
  week: { start: string; end: string } = manilaWeekWindow()
) {
  const inWeek = entriesInDateRange(entries, week.start, week.end)
  return {
    posRemittanceIn: inWeek
      .filter((entry) => entry.type === 'pos_remittance')
      .reduce((sum, entry) => sum + entry.amount, 0),
    expensesOut: inWeek
      .filter((entry) => entry.type === 'expense')
      .reduce((sum, entry) => sum + entry.amount, 0),
    ownerRemittance: inWeek
      .filter((entry) => entry.type === 'owner_remittance')
      .reduce((sum, entry) => sum + entry.amount, 0),
    auditorCashRemaining: walletRemaining(entries, 'auditor_cash'),
    pettyCashRemaining: walletRemaining(entries, 'petty_cash')
  }
}

function formatPeso(value: number): string {
  return `₱${value.toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ',')}`
}

function proposedRows(input: CreateLedgerEntryInput): Array<Pick<LedgerEntry, 'type' | 'wallet' | 'amount'>> {
  if (input.type === 'petty_issue') {
    return [
      { type: 'petty_issue', wallet: 'auditor_cash', amount: input.amount },
      { type: 'petty_issue', wallet: 'petty_cash', amount: input.amount }
    ]
  }
  return [{ type: input.type, wallet: input.wallet, amount: input.amount }]
}

export function validateEntry(
  input: CreateLedgerEntryInput,
  existing: LedgerEntry[]
): LedgerValidationResult {
  if (!(input.amount > 0)) {
    return { ok: false, message: 'Amount must be greater than 0.' }
  }
  if (input.type === 'expense' && !input.category) {
    return { ok: false, message: 'Category is required for expenses.' }
  }
  const auditorOnly = ['opening_cash', 'pos_float', 'pos_remittance', 'owner_remittance']
  if (auditorOnly.includes(input.type) && input.wallet !== 'auditor_cash') {
    return { ok: false, message: 'This entry must use auditor cash.' }
  }
  const next = [...existing]
  for (const row of proposedRows(input)) {
    const remaining = walletRemaining(next as LedgerEntry[], row.wallet === 'auditor_cash' ? 'auditor_cash' : 'petty_cash')
    const delta = entryDelta(row)
    const wallet = row.wallet
    if (remaining + delta[wallet] < 0) {
      const label = wallet === 'auditor_cash' ? 'Auditor cash' : 'Petty cash'
      return {
        ok: false,
        message: `${label} remaining is ${formatPeso(remaining)}. This would go negative.`
      }
    }
    next.push({
      id: 'proposed',
      entry_date: input.entry_date,
      type: row.type,
      wallet: row.wallet,
      amount: row.amount,
      category: input.category ?? null,
      note: null,
      transfer_id: null,
      created_by: null,
      created_at: '',
      updated_at: ''
    })
  }
  return { ok: true }
}

export function ledgerTypeLabel(type: LedgerEntryType): string {
  const labels: Record<LedgerEntryType, string> = {
    opening_cash: 'Starting cash',
    pos_float: 'Cash to POS',
    pos_remittance: 'Remittance from POS',
    owner_remittance: 'Remit to owner',
    petty_issue: 'Petty cash issued',
    expense: 'Expense'
  }
  return labels[type]
}

```ts
import { supabase } from '../supabase'
import {
  LEDGER_TABLE,
  type CreateLedgerEntryInput,
  type LedgerEntry
} from '../models/Ledger'
import { validateEntry } from '../modules/ledger/ledgerUtils'

const SELECT_COLUMNS =
  'id, entry_date, type, wallet, amount, category, note, transfer_id, created_by, created_at, updated_at'

function assertOk(input: CreateLedgerEntryInput, existing: LedgerEntry[]) {
  const result = validateEntry(input, existing)
  if (!result.ok) {
    throw new Error(result.message)
  }
}

function toInsertRow(input: CreateLedgerEntryInput, transferId: string | null) {
  return {
    entry_date: input.entry_date,
    type: input.type,
    wallet: input.wallet,
    amount: input.amount,
    category: input.type === 'expense' ? input.category ?? null : null,
    note: input.note ?? null,
    transfer_id: transferId,
    created_by: input.created_by ?? null
  }
}

export const ledgerService = {
  async list(): Promise<LedgerEntry[]> {
    const { data, error } = await supabase
      .from(LEDGER_TABLE)
      .select(SELECT_COLUMNS)
      .order('entry_date', { ascending: false })
      .order('created_at', { ascending: false })

    if (error) throw error
    return (data ?? []) as LedgerEntry[]
  },

  async create(input: CreateLedgerEntryInput, existing: LedgerEntry[]): Promise<LedgerEntry[]> {
    assertOk(input, existing)
    const rows =
      input.type === 'petty_issue'
        ? (() => {
            const transferId = crypto.randomUUID()
            return [
              toInsertRow({ ...input, wallet: 'auditor_cash' }, transferId),
              toInsertRow({ ...input, type: 'petty_issue', wallet: 'petty_cash' }, transferId)
            ]
          })()
        : [toInsertRow(input, null)]

    const { data, error } = await supabase.from(LEDGER_TABLE).insert(rows).select(SELECT_COLUMNS)
    if (error) throw error
    return (data ?? []) as LedgerEntry[]
  },

  async update(
    id: string,
    patch: Partial<CreateLedgerEntryInput>,
    existing: LedgerEntry[]
  ): Promise<LedgerEntry[]> {
    const current = existing.find((row) => row.id === id)
    if (!current) throw new Error('Ledger entry not found.')
    const merged: CreateLedgerEntryInput = {
      entry_date: patch.entry_date ?? current.entry_date,
      type: patch.type ?? current.type,
      wallet: patch.wallet ?? current.wallet,
      amount: patch.amount ?? current.amount,
      category: patch.category !== undefined ? patch.category : current.category,
      note: patch.note !== undefined ? patch.note : current.note,
      created_by: current.created_by
    }
    const others = existing.filter((row) =>
      current.transfer_id
        ? row.transfer_id !== current.transfer_id
        : row.id !== current.id
    )
    assertOk(merged, others)

    const payload = {
      entry_date: merged.entry_date,
      amount: merged.amount,
      note: merged.note ?? null,
      category: merged.type === 'expense' ? merged.category ?? null : null,
      updated_at: new Date().toISOString()
    }

    const query = supabase.from(LEDGER_TABLE).update(payload)
    const { data, error } = current.transfer_id
      ? await query.eq('transfer_id', current.transfer_id).select(SELECT_COLUMNS)
      : await query.eq('id', id).select(SELECT_COLUMNS)
    if (error) throw error
    return (data ?? []) as LedgerEntry[]
  },

  async remove(id: string, existing: LedgerEntry[]): Promise<void> {
    const current = existing.find((row) => row.id === id)
    if (!current) throw new Error('Ledger entry not found.')
    const query = supabase.from(LEDGER_TABLE).delete()
    const { error } = current.transfer_id
      ? await query.eq('transfer_id', current.transfer_id)
      : await query.eq('id', id)
    if (error) throw error
  }
}
```

import { computed, onMounted, ref } from 'vue'
import LedgerEntryModal from './LedgerEntryModal.vue'
import { authService } from '../services/authService'
import { ledgerService } from '../services/ledgerService'
import type { CreateLedgerEntryInput, LedgerEntry, LedgerEntryType, LedgerWallet } from '../models/Ledger'
import {
  dailyRemaining,
  entriesInDateRange,
  entryDelta,
  ledgerTypeLabel,
  manilaWeekWindow,
  walletRemaining,
  weekSummary
} from '../modules/ledger/ledgerUtils'

const entries = ref<LedgerEntry[]>([])
const error = ref('')
const isLoading = ref(false)
const modalOpen = ref(false)
const modalMode = ref<'add' | 'remit-owner' | 'edit'>('add')
const editing = ref<LedgerEntry | null>(null)
const week = manilaWeekWindow()
const startDate = ref(week.start)
const endDate = ref(week.end)
const walletFilter = ref('')
const typeFilter = ref('')

const auditorRemaining = computed(() => walletRemaining(entries.value, 'auditor_cash'))
const pettyRemaining = computed(() => walletRemaining(entries.value, 'petty_cash'))
const summary = computed(() => weekSummary(entries.value, week))
const visibleRows = computed(() => {
  let rows = entriesInDateRange(entries.value, startDate.value || '0000-01-01', endDate.value || '9999-12-31')
  if (walletFilter.value) rows = rows.filter((row) => row.wallet === walletFilter.value)
  if (typeFilter.value) rows = rows.filter((row) => row.type === typeFilter.value)
  return rows.filter((row, index, list) => {
    if (row.type !== 'petty_issue' || !row.transfer_id) return true
    return list.findIndex((other) => other.transfer_id === row.transfer_id) === index
  })
})

async function load() {
  isLoading.value = true
  error.value = ''
  try {
    entries.value = await ledgerService.list()
  } catch {
    error.value = 'Could not load ledger. Refresh to try again.'
    entries.value = []
  } finally {
    isLoading.value = false
  }
}

function formatCurrency(value: number): string {
  return value.toFixed(2).replace(/\B(?=(\d{3})+(?!\d))/g, ',')
}

function signedAmount(row: LedgerEntry): string {
  const delta = entryDelta(row)
  const value = row.wallet === 'auditor_cash' ? delta.auditor_cash : delta.petty_cash
  const abs = formatCurrency(Math.abs(value))
  return value < 0 ? `-₱${abs}` : `₱${abs}`
}

function openAdd() {
  modalMode.value = 'add'
  editing.value = null
  modalOpen.value = true
}

function openRemit() {
  modalMode.value = 'remit-owner'
  editing.value = null
  modalOpen.value = true
}

function openEdit(row: LedgerEntry) {
  modalMode.value = 'edit'
  editing.value = row
  modalOpen.value = true
}

async function onSubmit(input: CreateLedgerEntryInput) {
  error.value = ''
  const payload = {
    ...input,
    created_by: authService.getCurrentUser()?.username ?? null
  }
  try {
    if (modalMode.value === 'edit' && editing.value) {
      await ledgerService.update(editing.value.id, payload, entries.value)
    } else {
      await ledgerService.create(payload, entries.value)
    }
    modalOpen.value = false
    await load()
  } catch (err) {
    error.value = err instanceof Error ? err.message : 'Failed to save ledger entry.'
  }
}

async function onDelete(row: LedgerEntry) {
  const label = ledgerTypeLabel(row.type)
  if (!confirm(`Delete ${label} of ₱${formatCurrency(row.amount)}?`)) return
  try {
    await ledgerService.remove(row.id, entries.value)
    await load()
  } catch (err) {
    error.value = err instanceof Error ? err.message : 'Failed to delete ledger entry.'
  }
}

onMounted(load)
